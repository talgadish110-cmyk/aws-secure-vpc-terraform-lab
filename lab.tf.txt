# 6. Security Groups (Hardening: Bastion & Web)
resource "aws_security_group" "bastion" {
  name        = "bastion-sg"
  description = "Allow SSH from allowed IP only"
  vpc_id      = aws_vpc.main_vpc.id

  ingress {
    description = "SSH from Admin"
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}

resource "aws_security_group" "web" {
  name        = "web-server-sg"
  description = "Web servers security group"
  vpc_id      = aws_vpc.main_vpc.id

  ingress {
    description     = "HTTP from anywhere"
    from_port       = 80
    to_port         = 80
    protocol        = "tcp"
    cidr_blocks     = ["0.0.0.0/0"]
  }

  ingress {
    description     = "SSH from Bastion only"
    from_port       = 22
    to_port         = 22
    protocol        = "tcp"
    security_groups = [aws_security_group.bastion.id]
  }

  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }
}

# 11. Launch Template for Web Servers in Private Subnet
resource "aws_launch_template" "web_lt" {
  name_prefix   = "web-server-template"
  image_id      = "ami-0c7217cdde317cfec"
  instance_type = "t3.micro"

  network_interfaces {
    associate_public_ip_address = false
    security_groups             = [aws_security_group.web.id]
  }

  user_data = base64encode(<<-EOF
              #!/bin/bash
              yum update -y
              yum install -y httpd stress
              systemctl start httpd
              systemctl enable httpd
              echo "<h1>Hello from Private Web Server!</h1>" > /var/www/html/index.html
              EOF
  )

  tags = {
    Name = "Web-Server-Template"
  }
}

# 12. Auto Scaling Group (ASG) across Private Subnets
resource "aws_autoscaling_group" "web_asg" {
  desired_capacity    = 2
  min_size            = 1
  max_size            = 4
  vpc_zone_identifier = [aws_subnet.private_1.id, aws_subnet.private_2.id]
  target_group_arns   = [aws_lb_target_group.web_tg.arn]

  launch_template {
    id      = aws_launch_template.web_lt.id
    version = "$Latest"
  }

  tag {
    key                 = "Name"
    value               = "ASG-Web-Server"
    propagate_at_launch = true
  }
}
